// Adds a Playfield of openDrum devices to the open project, the pads in the order of the stock TR-909 Playfield.
// Hats, crash and ride play the stock preset's samples.

    await openDAW.showInfo("openDrum Hybrid", "No project is open. Create or load one first.")
    return
}

const stock = (name: string, uuid: string, duration: number) => ({uuid, name, duration, bpm: 0, sample_rate: 44100})
const drums = project.addInstrumentUnit("Playfield", {label: "openDrum Hybrid"}).instrument
drums.addPad("OpenDrumKick", {
    label: "Kick", tune: 46.84, pitch: 26.68, sweep: 0.0106, drift: 2.04, decay: 0.708, punch: -0.477, shape: 0.352,

}, {note: 61})
